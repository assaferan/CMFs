def reformat_file(fname):
    f = open(fname)
    lines = f.readlines()
    f.close()
    types = 'text:integer[]:integer[]:integer[]:integer[]\n'
    if lines[1] != types:
        lines = lines[:1] + [types] + lines[1:]
    if lines[2] != '\n':
        lines = lines[:2] + ['\n'] + lines[2:]
    cur = ''.join(lines)
    cur = cur.replace('[', '{').replace(']','}')
    f = open(fname, "w")
    f.write(cur)
    f.close()
